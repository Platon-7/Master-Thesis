# Full pip freeze of the working `robometer_gpu` conda env on Snellius (2026-05-11).
#
# This env trained the LoRA bake-off and is what Robometer-FT (transformers 4.57)
# uses. Built on Python 3.10 + CUDA 12.8 torch wheels.
#
# Build on a fresh cluster:
#
#   # 1. Create env with conda or miniforge:
#   conda create -n robometer_gpu -c conda-forge --override-channels python=3.10 -y
#   conda activate robometer_gpu
#
#   # 2. Install CUDA 12.8 torch first (don't let pip resolve a different cuda):
#   pip install --index-url https://download.pytorch.org/whl/cu128 \
#       torch==2.8.0 torchvision==0.23.0
#
#   # 3. Then the rest:
#   pip install -r requirements-robometer_gpu.txt
#
#   # 4. Verify torch sees CUDA:
#   python -c "import torch; print(torch.__version__, torch.cuda.is_available())"
#
# Notes:
#   * `flash_attn` is NOT in this freeze — the env doesn't have it built. The
#     codebase falls back to SDPA when flash_attn import fails (see
#     setup_utils.py line 801). If you want flash_attn, build it on the cluster
#     with `MAX_JOBS=4 pip install --no-build-isolation flash-attn` on a node
#     with nvcc + CUDA toolkit (login nodes usually don't have nvcc).
#   * `tensorflow*` deps from pyproject.toml are NOT in this freeze either —
#     they're only needed for OXE dataset re-uploading, not training. Skip.
#   * `unsloth` is in here but the FT runs set `model.use_unsloth=false` so its
#     presence is harmless (just adds import surface).
#
# ---
